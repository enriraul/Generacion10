Comndos DOCKER
===========================

docker ps 
cd  path_to_file
docker cp db.sql postgres-clase:db.sql
docker cp data.sql postgres-clase:data.sql
docker exec -it postgres-clase bash
psql -U admin
CREATE DATABASE dreams WITH OWNER admin;
psql -U admin -d dreams -f db.sql
psql -U admin -d dreams -f data.sql
psql -U admin -d dreams 
select * from cliente;
select * from producto;
